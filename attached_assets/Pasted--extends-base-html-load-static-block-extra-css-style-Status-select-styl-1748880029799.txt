<!-- Status handling and fixes -->
<script>
document.addEventListener('DOMContentLoaded', function() {
    // Status color handling
    const statusSelect = document.getElementById('id_status');
    
    function updateStatusColor() {
        if (statusSelect) {
            const status = statusSelect.value;
            statusSelect.setAttribute('data-status', status);
        }
    }
    
    // Set initial color and update on change
    updateStatusColor();
    if (statusSelect) {
        statusSelect.addEventListener('change', function() {
            updateStatusColor();
            console.log('Status changed to:', this.value); // Debug log
        });
        
        // Make sure the select is not disabled
        statusSelect.disabled = false;
        statusSelect.removeAttribute('readonly');
    }
    
    // Fix GM% decimal places
    const gmField = document.getElementById('gm_percentage');
    if (gmField) {
        // Format existing value
        const currentValue = parseFloat(gmField.value);
        if (!isNaN(currentValue)) {
            gmField.value = currentValue.toFixed(1);
        }
        
        // Override any other scripts that might be setting the value
        const originalSetAttribute = gmField.setAttribute;
        gmField.setAttribute = function(name, value) {
            if (name === 'value') {
                const numValue = parseFloat(value);
                if (!isNaN(numValue)) {
                    value = numValue.toFixed(1);
                }
            }
            originalSetAttribute.call(this, name, value);
        };
    }
});
</script>
